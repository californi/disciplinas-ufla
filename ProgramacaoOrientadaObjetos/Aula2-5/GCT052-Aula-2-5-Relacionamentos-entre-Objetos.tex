% Seção 1: Introdução
\section{Introdução aos Relacionamentos}

\begin{frame}{O que são Relacionamentos entre Objetos?}
\begin{itemize}
\item \textbf{Relacionamentos} definem como objetos se conectam e interagem
\item \textbf{Associação}: Relacionamento simples entre objetos independentes
\item \textbf{Agregação}: Relacionamento "tem um" onde o objeto parte pode existir independentemente
\item \textbf{Composição}: Relacionamento "tem um" onde o objeto parte não pode existir sem o todo
\end{itemize}

\vspace{0.5cm}
\textbf{Analogia do Mundo Real:}
\begin{itemize}
\item \textbf{Associação}: Pessoa e Livro (pessoa pode ler livro)
\item \textbf{Agregação}: Universidade e Professor (professor pode existir sem universidade)
\item \textbf{Composição}: Casa e Quarto (quarto não existe sem casa)
\end{itemize}
\end{frame}

\begin{frame}{Por que estudar Relacionamentos?}
\begin{enumerate}
\item \textbf{Modelagem Realista}: Representa melhor o mundo real
\item \textbf{Reutilização}: Objetos podem ser reutilizados em diferentes contextos
\item \textbf{Flexibilidade}: Facilita mudanças e manutenção
\item \textbf{Organização}: Código mais estruturado e compreensível
\item \textbf{Design Patterns}: Base para padrões de projeto avançados
\end{enumerate}

\vspace{0.5cm}
\textbf{Vantagens:}
\begin{itemize}
\item Código mais modular e flexível
\item Facilita trabalho em equipe
\item Melhora manutenibilidade
\item Permite modelagem complexa
\end{itemize}
\end{frame}

% Seção 2: Associação
\section{Associação}

\begin{frame}{O que é Associação?}
\begin{itemize}
\item \textbf{Associação}: Relacionamento mais fraco entre objetos independentes
\item \textbf{Características}:
\begin{itemize}
\item Objetos podem existir independentemente
\item Relacionamento temporário ou permanente
\item Ambos os objetos têm vida própria
\item Comunicação através de métodos
\end{itemize}
\end{itemize}

\vspace{0.3cm}
\textbf{Exemplos do Dia a Dia:}
\begin{itemize}
\item \textbf{Pessoa} e \textbf{Carro} (pessoa pode dirigir carro)
\item \textbf{Professor} e \textbf{Aluno} (professor ensina aluno)
\item \textbf{Cliente} e \textbf{Produto} (cliente compra produto)
\end{itemize}
\end{frame}

\begin{frame}{Tipos de Associação}
\begin{enumerate}
\item \textbf{Associação Simples}: Relacionamento básico entre duas classes
\item \textbf{Associação Unidirecional}: Um objeto conhece o outro, mas não vice-versa
\item \textbf{Associação Bidirecional}: Ambos os objetos se conhecem mutuamente
\item \textbf{Associação Reflexiva}: Objeto se relaciona consigo mesmo
\end{enumerate}

\vspace{0.5cm}
\textbf{Multiplicidade:}
\begin{itemize}
\item \textbf{1:1}: Um para um
\item \textbf{1:N}: Um para muitos
\item \textbf{N:N}: Muitos para muitos
\end{itemize}
\end{frame}

% Seção 3: Agregação
\section{Agregação}

\begin{frame}{O que é Agregação?}
\begin{itemize}
\item \textbf{Agregação}: Relacionamento "tem um" onde a parte pode existir independentemente
\item \textbf{Características}:
\begin{itemize}
\item Objeto todo contém objetos partes
\item Partes podem existir sem o todo
\item Relacionamento de propriedade fraca
\item Destruição do todo não destrói as partes
\end{itemize}
\end{itemize}

\vspace{0.3cm}
\textbf{Exemplos do Mundo Real:}
\begin{itemize}
\item \textbf{Universidade} e \textbf{Professor} (universidade tem professores)
\item \textbf{Carro} e \textbf{Roda} (carro tem rodas)
\item \textbf{Escola} e \textbf{Aluno} (escola tem alunos)
\end{itemize}
\end{frame}

\begin{frame}{Características da Agregação}
\begin{itemize}
\item \textbf{Símbolo}: Losango vazio na UML
\item \textbf{Propriedade}: "tem um" ou "contém"
\item \textbf{Vida Útil}: Partes podem sobreviver ao todo
\item \textbf{Criação}: Todo pode criar ou receber partes
\item \textbf{Destruição}: Todo pode ser destruído sem afetar partes
\end{itemize}

\vspace{0.5cm}
\textbf{Implementação em C\#:}
\begin{itemize}
\item Uso de propriedades ou campos
\item Injeção de dependência
\item Coleções para múltiplas partes
\item Métodos para gerenciar partes
\end{itemize}
\end{frame}

% Seção 4: Composição
\section{Composição}

\begin{frame}{O que é Composição?}
\begin{itemize}
\item \textbf{Composição}: Relacionamento "tem um" onde a parte não pode existir sem o todo
\item \textbf{Características}:
\begin{itemize}
\item Objeto todo possui objetos partes
\item Partes não podem existir independentemente
\item Relacionamento de propriedade forte
\item Destruição do todo destrói as partes
\end{itemize}
\end{itemize}

\vspace{0.3cm}
\textbf{Exemplos do Mundo Real:}
\begin{itemize}
\item \textbf{Casa} e \textbf{Quarto} (casa tem quartos)
\item \textbf{Carro} e \textbf{Motor} (carro tem motor)
\item \textbf{Pessoa} e \textbf{Coração} (pessoa tem coração)
\end{itemize}
\end{frame}

\begin{frame}{Características da Composição}
\begin{itemize}
\item \textbf{Símbolo}: Losango preenchido na UML
\item \textbf{Propriedade}: "possui" ou "é composto por"
\item \textbf{Vida Útil}: Partes dependem do todo
\item \textbf{Criação}: Todo cria suas próprias partes
\item \textbf{Destruição}: Todo destrói suas partes ao ser destruído
\end{itemize}

\vspace{0.5cm}
\textbf{Implementação em C\#:}
\begin{itemize}
\item Criação de objetos no construtor
\item Propriedades somente leitura
\item Gerenciamento de ciclo de vida
\item Métodos para operações nas partes
\end{itemize}
\end{frame}

% Seção 5: Comparação dos Relacionamentos
\section{Comparação dos Relacionamentos}

\begin{frame}{Diferenças entre os Relacionamentos}
\begin{center}
\begin{tabular}{|l|c|c|c|}
\hline
\textbf{Característica} & \textbf{Associação} & \textbf{Agregação} & \textbf{Composição} \\
\hline
Força do relacionamento & Fraca & Média & Forte \\
\hline
Existência independente & Sim & Sim & Não \\
\hline
Propriedade & Não & Parcial & Total \\
\hline
Ciclo de vida & Independente & Independente & Dependente \\
\hline
Símbolo UML & Linha simples & Losango vazio & Losango preenchido \\
\hline
\end{tabular}
\end{center}

\vspace{0.5cm}
\textbf{Regra Prática:}
\begin{itemize}
\item \textbf{Associação}: "usa" ou "conhece"
\item \textbf{Agregação}: "tem" (pode existir sem)
\item \textbf{Composição}: "possui" (não pode existir sem)
\end{itemize}
\end{frame}

\begin{frame}{Exemplos Práticos de Comparação}
\textbf{Cenário: Sistema de Biblioteca}

\begin{itemize}
\item \textbf{Associação}: \texttt{Bibliotecario} e \texttt{Livro}
\begin{itemize}
\item Bibliotecário pode trabalhar com diferentes livros
\item Livro pode ser usado por diferentes bibliotecários
\end{itemize}

\item \textbf{Agregação}: \texttt{Biblioteca} e \texttt{Livro}
\begin{itemize}
\item Biblioteca contém livros
\item Livro pode existir fora da biblioteca
\end{itemize}

\item \textbf{Composição}: \texttt{Livro} e \textbf{Página}
\begin{itemize}
\item Livro é composto por páginas
\item Página não pode existir sem o livro
\end{itemize}
\end{itemize}
\end{frame}

% Seção 9: Boas Práticas
\section{Boas Práticas}

\begin{frame}{Design de Relacionamentos}
\begin{itemize}
\item \textbf{Escolha o relacionamento correto} baseado no domínio
\item \textbf{Use composição} quando a parte não pode existir sem o todo
\item \textbf{Use agregação} quando a parte pode existir independentemente
\item \textbf{Use associação} para relacionamentos temporários ou de uso
\end{itemize}

\vspace{0.5cm}
\textbf{Princípios:}
\begin{itemize}
\item \textbf{Princípio da Responsabilidade Única}: Cada classe tem uma responsabilidade
\item \textbf{Princípio Aberto/Fechado}: Aberto para extensão, fechado para modificação
\item \textbf{Princípio da Inversão de Dependência}: Dependa de abstrações, não de concretizações
\end{itemize}
\end{frame}

\begin{frame}{Implementação em C\#}
\begin{itemize}
\item \textbf{Use properties} para relacionamentos simples
\item \textbf{Use collections} para relacionamentos 1:N
\item \textbf{Implemente métodos} para operações relacionadas
\item \textbf{Use construtores} apropriados para cada tipo
\item \textbf{Documente} os relacionamentos com XML comments
\end{itemize}

\vspace{0.5cm}
\textbf{Exemplo de XML Comments:}
\begin{itemize}
\item \texttt{/// <summary>}
\item \texttt{/// Relacionamento de agregação com professores}
\item \texttt{/// </summary>}
\end{itemize}
\end{frame}

\begin{frame}{Testes e Validação}
\begin{itemize}
\item \textbf{Teste a criação} de objetos relacionados
\item \textbf{Valide a destruição} apropriada
\item \textbf{Verifique a independência} quando aplicável
\item \textbf{Teste operações} em relacionamentos
\end{itemize}

\vspace{0.5cm}
\textbf{Validações importantes:}
\begin{itemize}
\item Objetos não podem ser nulos quando necessário
\item Relacionamentos devem ser consistentes
\item Ciclo de vida deve ser respeitado
\item Operações devem manter integridade
\end{itemize}
\end{frame}
