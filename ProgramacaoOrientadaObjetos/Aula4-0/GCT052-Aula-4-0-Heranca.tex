% Seção 1: Introdução
\section{Introdução à Herança}

\begin{frame}{O que é Herança?}
\begin{itemize}
\item \textbf{Herança}: Mecanismo que permite criar novas classes baseadas em classes existentes
\item \textbf{Classe Base (Pai)}: Classe que fornece características e comportamentos
\item \textbf{Classe Derivada (Filha)}: Classe que herda e pode estender ou modificar
\item \textbf{Relacionamento "é um"}: Classe filha é um tipo especial da classe pai
\end{itemize}

\vspace{0.5cm}
\textbf{Analogia do Mundo Real:}
\begin{itemize}
\item \textbf{Classe Base}: Animal
\item \textbf{Classes Derivadas}: Cachorro, Gato, Pássaro
\item \textbf{Relacionamento}: Cachorro é um Animal, Gato é um Animal
\end{itemize}
\end{frame}

\begin{frame}{Por que usar Herança?}
\begin{enumerate}
\item \textbf{Reutilização de Código}: Evita duplicação
\item \textbf{Organização}: Hierarquia clara e lógica
\item \textbf{Manutenção}: Mudanças na classe base afetam todas as derivadas
\item \textbf{Polimorfismo}: Permite tratar objetos de forma uniforme
\item \textbf{Extensibilidade}: Fácil adicionar novos tipos
\end{enumerate}

\vspace{0.5cm}
\textbf{Vantagens:}
\begin{itemize}
\item Código mais limpo e organizado
\item Facilita manutenção e evolução
\item Permite modelagem hierárquica do mundo real
\item Base para polimorfismo
\end{itemize}
\end{frame}

% Seção 2: Conceitos Fundamentais
\section{Conceitos Fundamentais}

\begin{frame}{Terminologia da Herança}
\begin{itemize}
\item \textbf{Classe Base (Superclasse)}: Classe que é herdada
\item \textbf{Classe Derivada (Subclasse)}: Classe que herda
\item \textbf{Herança Simples}: Uma classe pode herdar de apenas uma classe base
\item \textbf{Herança Múltipla}: Não suportada em C\# (mas interfaces sim)
\item \textbf{Modificador protected}: Acessível na classe e nas derivadas
\end{itemize}

\vspace{0.3cm}
\textbf{Exemplo:}
\begin{itemize}
\item \textbf{Classe Base}: \texttt{Veiculo}
\item \textbf{Classe Derivada}: \texttt{Carro : Veiculo}
\item \textbf{Relacionamento}: Carro é um Veículo
\end{itemize}
\end{frame}

\begin{frame}{Sintaxe de Herança em C\#}
\textbf{Sintaxe básica:}
\begin{itemize}
\item \texttt{public class ClasseDerivada : ClasseBase}
\item \texttt{public class Cachorro : Animal}
\item \texttt{public class Carro : Veiculo}
\end{itemize}

\vspace{0.5cm}
\textbf{Características:}
\begin{itemize}
\item Usa dois pontos (\texttt{:}) para indicar herança
\item Classe derivada herda todos os membros públicos e protegidos
\item Pode adicionar novos membros
\item Pode sobrescrever métodos virtuais
\item Pode chamar construtor da classe base com \texttt{base()}
\end{itemize}
\end{frame}

% Seção 3: Modificadores de Acesso
\section{Modificadores de Acesso}

\begin{frame}{Modificadores e Herança}
\begin{enumerate}
\item \textbf{public}: Acessível de qualquer lugar
\item \textbf{private}: Acessível apenas na própria classe
\item \textbf{protected}: Acessível na classe e nas classes derivadas
\item \textbf{internal}: Acessível no mesmo assembly
\item \textbf{protected internal}: Acessível no assembly ou classes derivadas
\end{enumerate}

\vspace{0.5cm}
\textbf{Boas Práticas:}
\begin{itemize}
\item Use \textbf{protected} para membros que classes derivadas precisam acessar
\item Mantenha encapsulamento com \textbf{private}
\item Use \textbf{public} para interface pública
\end{itemize}
\end{frame}

\begin{frame}{Construtores e Herança}
\begin{itemize}
\item \textbf{Construtor da Base}: Pode ser chamado com \texttt{base()}
\item \textbf{Ordem de Execução}: Base primeiro, depois derivada
\item \textbf{Construtor Padrão}: Se não especificar, chama construtor sem parâmetros da base
\item \textbf{Construtor Específico}: Deve chamar explicitamente com \texttt{base()}
\end{itemize}

\vspace{0.3cm}
\textbf{Exemplo:}
\begin{itemize}
\item \texttt{public Cachorro(string nome) : base(nome, "Cachorro")}
\item Primeiro executa construtor de \texttt{Animal}
\item Depois executa construtor de \texttt{Cachorro}
\end{itemize}
\end{frame}

% Seção 4: Sobrescrita de Métodos
\section{Sobrescrita de Métodos}

\begin{frame}{Métodos Virtuais e Override}
\begin{itemize}
\item \textbf{virtual}: Permite que método seja sobrescrito
\item \textbf{override}: Sobrescreve método virtual da classe base
\item \textbf{sealed}: Impede que método seja sobrescrito novamente
\item \textbf{new}: Oculta membro da classe base (não recomendado)
\end{itemize}

\vspace{0.5cm}
\textbf{Exemplo:}
\begin{itemize}
\item \texttt{public virtual void FazerSom()} na classe base
\item \texttt{public override void FazerSom()} na classe derivada
\item Cada classe pode ter sua própria implementação
\end{itemize}
\end{frame}

\begin{frame}{Polimorfismo}
\begin{itemize}
\item \textbf{Polimorfismo}: Capacidade de objetos diferentes responderem de forma diferente à mesma mensagem
\item \textbf{Ligação Tardia}: Decisão sobre qual método chamar é feita em tempo de execução
\item \textbf{Referência da Base}: Pode apontar para objeto da derivada
\item \textbf{Comportamento Específico}: Método sobrescrito é chamado
\end{itemize}

\vspace{0.3cm}
\textbf{Exemplo:}
\begin{itemize}
\item \texttt{Animal animal = new Cachorro();}
\item \texttt{animal.FazerSom();} // Chama método de Cachorro
\item Mesma referência, comportamentos diferentes
\end{itemize}
\end{frame}

% Seção 6: Implementação em C#
\section{Implementação em C\#}

\begin{frame}{Estrutura Básica de Herança}
\textbf{Elementos principais:}
\begin{itemize}
\item \textbf{Classe Base}: Define membros comuns
\item \textbf{Classe Derivada}: Herda e estende funcionalidades
\item \textbf{Construtores}: Chamam construtor da base com \texttt{base()}
\item \textbf{Métodos Virtuais}: Permitem sobrescrita
\item \textbf{Métodos Override}: Implementam comportamento específico
\end{itemize}

\vspace{0.3cm}
\textbf{Exemplo completo:} Ver arquivos \texttt{Exemplos/}
\end{frame}

\begin{frame}{Padrões de Implementação}
\textbf{Classe Base:}
\begin{itemize}
\item Atributos protegidos ou privados
\item Métodos virtuais para comportamentos variáveis
\item Construtores para inicialização
\item Métodos comuns a todas as derivadas
\end{itemize}

\textbf{Classe Derivada:}
\begin{itemize}
\item Herda membros da base
\item Adiciona atributos específicos
\item Sobrescreve métodos virtuais
\item Chama construtor da base
\end{itemize}
\end{frame}

% Seção 9: Boas Práticas
\section{Boas Práticas}

\begin{frame}{Design de Herança}
\begin{itemize}
\item \textbf{Use herança para relacionamento "é um"}
\item \textbf{Evite herança profunda} (máximo 3-4 níveis)
\item \textbf{Prefer composição sobre herança} quando apropriado
\item \textbf{Use classes abstratas} para classes base que não devem ser instanciadas
\item \textbf{Documente} a hierarquia de herança
\end{itemize}

\vspace{0.5cm}
\textbf{Princípios:}
\begin{itemize}
\item \textbf{Princípio da Substituição de Liskov}: Objetos derivados devem poder substituir objetos base
\item \textbf{Open/Closed Principle}: Aberto para extensão, fechado para modificação
\item \textbf{DRY}: Don't Repeat Yourself - reutilize código
\end{itemize}
\end{frame}

\begin{frame}{Implementação em C\#}
\begin{itemize}
\item \textbf{Use \texttt{virtual}} para métodos que podem ser sobrescritos
\item \textbf{Use \texttt{override}} para sobrescrever métodos virtuais
\item \textbf{Use \texttt{base()}} para chamar construtor da base
\item \textbf{Use \texttt{protected}} para membros acessíveis às derivadas
\item \textbf{Documente} com XML comments
\end{itemize}

\vspace{0.5cm}
\textbf{Evite:}
\begin{itemize}
\item Herança apenas para reutilizar código
\item Classes base muito grandes
\item Hierarquias muito profundas
\item Uso de \texttt{new} para ocultar membros
\end{itemize}
\end{frame}

\begin{frame}{Testes e Validação}
\begin{itemize}
\item \textbf{Teste} cada classe derivada independentemente
\item \textbf{Valide} que métodos sobrescritos funcionam corretamente
\item \textbf{Teste} polimorfismo com referências da base
\item \textbf{Verifique} que construtores são chamados na ordem correta
\end{itemize}

\vspace{0.5cm}
\textbf{Validações importantes:}
\begin{itemize}
\item Herança mantém comportamento esperado
\item Sobrescrita funciona corretamente
\item Polimorfismo se comporta como esperado
\item Encapsulamento é respeitado
\end{itemize}
\end{frame}
